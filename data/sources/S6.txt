// pri函数定义，用于判断一个整数是否为素数

int pri(int j)

{

    // 如果j小于等于1，直接返回0，因为1及以下的数不是素数

    if (j <= 1)

    {

        return 0;

    }

    // 如果j小于等于3，返回1，2和3是素数

    if (j <= 3)

    {

        return 1;

    }

    // 如果j能被2或3整除，说明j不是素数，返回0

    if (j % 2 == 0 || j % 3 == 0)

    {

        return 0;

    }

    // 初始化i为5，用于后续的循环判断

    int i = 5;

    // 当i的平方小于等于j时，继续循环

    while (i * i <= j) {

        // 如果j能被i或i + 2整除，说明j不是素数，返回0

        if (j % i == 0 || j % (i + 2) == 0)

        {

            return 0;

        }

        // i每次增加6，因为大于3的素数都可以表示为6k ± 1的形式

        i += 6;

    }

    // 如果循环结束没有返回0，说明j是素数，返回1

    return 1;

}
